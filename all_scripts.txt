"""
scripts/train.py

Shared training loop for all four attention variants

Loads the config for the specified task, builds the model with the specified attention variant, trains on the LRA dataset and saves the trained model and training log to results/.

Usage:
    python3 scipt/evaluate.py --variant baseline --task listops
    python3 scipt/evaluate.py --variant longformer --task text_classification
    python3 scipt/evaluate.py --variant linformer --task lisops
    python3 scipt/evaluate.py --variant reformer --task text_classificaiton

All four variants run through the same BaselineTransformerEncoder stack - only the attention class changes, satisfying the unified evaluation framework.
"""

import os
import sys
import json
import argparse
import yaml
import torch
import torch.nn as nn
import torch.optim as optim

sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

from baseline.attention import(BaselineTransformerEncoder, MultiHeadSelfAttention)
from longformer.attention import LongformerAttention
from linformer.attention import LinformerAttention
from reformer.attention import ReformerAttention

def load_config(task):
    config_path = os.path.join(os.path.dirname(os.path.dirname(os.path.abspath(__file__))), "configs", f"{task}.yaml")
    with open(config_path, "r") as f:
        return yaml.safe_load(f)
    
def build_model(config, variant):
    """
    Build the encoder with the specified attention variant.

    All four variants use the same BaselineTransformerEncoder - only attn_cls changes.
    Variant specified kwargs (keyword arguments) are passed via a closure so the encoder block's attn_cls(d_model, num_heads, dropout) call still works for all varaints.
    """
    if variant == "baseline":
        attn_cls = MultiHeadSelfAttention

    elif variant == "longformer":
        window_size = config["longformer_window_size"]
        def attn_cls(d_model, num_heads, dropout):
            return LongformerAttention(d_model, num_heads, dropout=dropout, window_size=window_size)
        
    elif variant == "linformer":
        proj_dim = config["linformer_proj_dim"]
        seq_len = config["seq_len"]
        def attn_cls(d_model, num_heads, dropout):
            return LinformerAttention(d_model, num_heads, dropout=dropout, seq_len=seq_len, proj_dim=proj_dim)
        
    elif variant == "reformer":
        n_rounds = config["reformer_n_rounds"]
        chunk_size = config["reformer_chunk_size"]
        n_buckets = config["reformer_n_buckets"]
        def attn_cls(d_model, num_heads, dropout):
            return ReformerAttention(d_model, num_heads, dropout=dropout, n_rounds=n_rounds, chunk_size=chunk_size, n_buckets=n_buckets)
        
    else:
        raise ValueError(f"unknown variant: {variant}")
    
    model = BaselineTransformerEncoder(
        vocab_size=config["vocab_size"],
        d_model=config["d_model"],
        num_heads=config["num_heads"],
        num_layers=config["num_layers"],
        num_classes=config["num_classes"],
        max_seq_len=config["seq_len"],
        d_ff=config["d_ff"],
        dropout=config["dropout"],
        attn_cls=attn_cls,
    )

    return model

def get_device():
    if torch.cuda.is_available():
        return torch.device("cuda")
    elif torch.backends.mps.is_available():
        return torch.device("mps") # apple silicon
    else:
        return torch.debice("cpu")
    
def load_data(config):
    """
    Load the LRA dataset for the specified task using tensorflow_datasets.
    Returns train and test DataLoaders. The LRA datasets are loaded as described in [X] Tay et al. (2020).
    """
    import tensorflow_datasets as tfds
    from torch.utils.data import DataLoader, TensorDataset

    task = config["task"]
    seq_len = config["seq_len"]
    batch_size = config["batch_size"]

    if task == "listops":
        # listops is not directly in tfds - load from local data directory
        # runs scripts/download_data.py first to generate the dataset
        data_dir = os.path.join(os.path.dirname(os.path.dirname(os.path.abspath(__file__))), "data", "listops")
        if not os.path.exists(data_dir):
            raise FileNotFoundError(f"ListOps data not founds at {data_dir}. Run scripts/download_data.py first")
        train_data = torch.load(os.path.join(data_dir, "train.pt"), weights_only=False)
        test_data = torch.load(os.path.join(data_dir, "test.pt"), weights_only=False)

    elif task == "text_classification":
        # imdb is available directly via tfds
        ds_train = tfds.load("imdb_reviews/bytes", split="train", as_supervised=True)
        ds_test = tfds.load("imdb_reviews/bytes", split="test", as_supervised=True)

        def process(ds):
            input_ids_list, labels_list, masks_list = [], [], []
            for text, label in tfds.as_numpy(ds):
                tokens = text[:seq_len]
                pad_len = seq_len - len(tokens)
                mask = [1] * len(tokens) + [0] * pad_len
                tokens = list(tokens) + [0] * pad_len
                input_ids_list.append(tokens)
                labels_list.append(int(label))
                masks_list.append(mask)
            return TensorDataset(
                torch.tensor(input_ids_list, dtype=torch.long),
                torch.tensor(labels_list, dtype=torch.long),
                torch.tensor(masks_list, dtype=torch.long),
            )
        
        train_data = process(ds_train)
        test_data = process(ds_test)

    train_loader = DataLoader(train_data, batch_size=batch_size, shuffle=True)
    test_loader = DataLoader(test_data, batch_size=batch_size, shuffle=False)

    return train_loader, test_loader

def train_epoch(model, loader, optimizer, criterion, device):
    model.train()
    total_loss = 0
    correct = 0
    total = 0

    for batch in loader:
        input_ids, labels, attention_mask = batch
        input_ids = input_ids.to(device)
        labels = labels.to(device)
        attention_mask = attention_mask.to(device)

        optimizer.zero_grad()
        out = model(input_ids, attention_mask=attention_mask)
        loss = criterion(out["logits"], labels)
        loss.backward()
        optimizer.step()

        total_loss += loss.item()
        preds = out["logits"].argmax(dim=-1)
        correct += (preds == labels).sum().item()
        total += labels.size(0)

    return total_loss / len(loader), correct / total

def eval_epoch(model, loader, criterion, device):
    model.eval()
    total_loss = 0
    correct = 0
    total = 0

    with torch.no_grad():
        for batch in loader:
            input_ids, labels, attention_mask = batch
            input_ids = input_ids.to(device)
            labels = labels.to(device)
            attention_mask = attention_mask.to(device)

            out = model(input_ids, attention_mask=attention_mask)
            loss = criterion(out["logits"], labels)

            total_loss += loss.item()
            preds = out["logits"].argmax(dim=-1)
            correct += (preds == labels).sum().item()
            total += labels.size(0)

    return total_loss / len(loader), correct / total

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--variant", required=True, choices=["baseline", "longformer", "linformer", "reformer"])
    parser.add_argument("--task", required=True, choices=["listops", "text_classification", "integration_test"])
    args = parser.parse_args()

    config = load_config(args.task)
    torch.manual_seed(config["seed"]) # fix seed for reproducibility

    device = get_device()
    print(f"using device: {device}")

    print(f"loading data for {args.task}..")
    train_loader, test_loader = load_data(config)

    print(f"building {args.variant} model..")
    model = build_model(config, args.variant)
    model = model.to(device)

    n_params = sum(p.numel() for p in model.parameters() if p.requires_grad)
    print(f"parameters: {n_params:,}")

    criterion = nn.CrossEntropyLoss()
    optimizer = optim.Adam(model.parameters(), lr=config["learning_rate"])

    # linear warmup scheduler
    def lr_lambda(step):
        if step < config["warmup_steps"]:
            return step / config["warmup_steps"]
        return 1.0
    
    scheduler = optim.lr_scheduler.LambdaLR(optimizer, lr_lambda)

    log = []
    best_acc = 0.0

    for epoch in range(config["epochs"]):
        train_loss, train_acc = train_epoch(model, train_loader, optimizer, criterion, device)
        test_loss, test_acc = eval_epoch(model, test_loader, criterion, device)
        scheduler.step()

        print(f"epoch {epoch+1}/{config['epochs']} - train loss: {train_loss:.4f}, train acc: {train_acc:.4f}, test loss: {test_loss:.4f}, test acc: {test_acc:.4f}")

        log.append({"epoch": epoch + 1, "train_loss": train_loss, "train_acc": train_acc, "test_loss": test_loss, "test_acc": test_acc})

        # save best model
        if test_acc > best_acc:
            best_acc = test_acc
            results_dir = os.path.join(os.path.dirname(os.path.dirname(os.path.abspath(__file__))), "results")
            os.makedirs(results_dir, exist_ok=True)
            torch.save(model.state_dict(), os.path.join(results_dir, f"{args.variant}_{args.task}_best.pt"))

    results_dir = os.path.join(os.path.dirname(os.path.dirname(os.path.abspath(__file__))), "results")
    os.makedirs(results_dir, exist_ok=True)
    log_path = os.path.join(results_dir, f"{args.variant}_{args.task}_log.json")
    with open(log_path, "w") as f:
        json.dump(log, f, indent=2)

    print(f"\ntraining complete. best test accuracy: {best_acc:.4f}")
    print(f"log saved to {log_path}")


if __name__ == "__main__":
    main()"""

def measure_inference(model, input_ids, attention_mask, device, n_runs=10):
    """
    Measure average inference time and peak memory over n_runs forward passes.

    Runs a warmup pass first to avoid measuring JIT compilation overhead then averages over n_runs to get a stable estimate.

    Args:
        model: the trained model in eval mode.
        input_ids (Tensor): (batch, num_tokens)
        attention_mask (Tensor): (batch, num_tokens)
        device: torch device
        n_runs (int): number of forward passes to average over

    Returns:
        avg_time_ms (float): average forward pass time in milliseconds
        peak_memory_mb (float): peak memory allocated in MB (GPU only, 0 on CPU)
    """
    model.eval()
    input_ids = input_ids.to(device)
    attention_mask = attention_mask.to(device)

    # warmup pass
    with torch.no_grad():
        _ = model(input_ids, attention_mask=attention_mask)

    # reset memory stats before measuring
    if device.type == "cuda":
        torch.cuda.reset_peak_memory_stats()
    elif device.type == "mps":
        torch.mps.empty_cache()

    times = []
    with torch.no_grad():
        for _ in range(n_runs):
            if device.type == "cuda":
                start = torch.cuda.Event(enable_timing=True)
                end = torch.cuda.Event(enable_timing=True)
                start.record()
                _ = model(input_ids, attention_mask=attention_mask)
                end.record()
                torch.cuda.synchronize()
                times.append(start.elapsed_time(end))
            else:
                t0 = time.perf_counter()
                _ = model(input_ids, attention_mask=attention_mask)
                t1 = time.perf_counter()
                times.append((t1 - t0) * 1000) # convert to ms

    avg_time_ms = sum(times) / len(times)

    # peak memory - only available on CUDA
    if device.type == "cuda":
        peak_memory_mb = torch.cuda.max_memory_allocated() / 10224 ** 2
    else:
        # mps and cpu dont expose peak memory in the same way
        peak_memory_mb = 0.0

    return avg_time_ms, peak_memory_mb

def evaluate(model, loader, device):
    """
    Run the model on the full test set and compute accuracy.

    Args:
        model: tained model in eval mode
        loader: test Dataloader
        device: torch device

    Returns:
        accuracy (float): fraction of correct predictions
    """
    model.eval()
    correct = 0
    total = 0

    with torch.no_grad():
        for batch in loader:
            inputs_ids, labels, attention_mask = batch
            input_ids = input_ids.to(device)
            labels = labels.to(device)
            attention_mask = attention_mask.to(device)

            out = model(input_ids, attention_mask=attention_mask)
            preds = out["logits"].argmax(dim=-1)
            correct += (preds == labels).sum().item()
            total += labels.size(0)

    return correct / total

    parser.add_argument("--task", required=True, choices=["listops", "text_classification", "integration_test"])
    parser.add_argument("--n_runs", type=int, default=10, help="number of forward passes to average for timing")
    args = parser.parse_args()

    config = load_config(args.task)
    print(f"using device: {device}")

    results_dir = os.path.join(os.path.dirname(os.path.dirname(os.path.abspath(__file__))), "results")

    model = build_model(config, args.variant)
    model.load_state_dict(torch.load(model_path, map_location=device))
    model = model.to(device)
    model.eval()

    print(f"loading test data for {args.task}...")
    _, test_loader = load_data(config)

    print("evaluating accuracy..")
    accuracy = evaluate(model, test_loader, device)
    print(f"accuracy: {accuracy:.4f}")

    # use the first batch for timing - representative of a single forward pass
    first_batch - next(iter(test_loader))
    input_ids, _, attention_mask = first_batch

    print(f"benchamrking inference time over {args.n_runs} runs...")
    avg_time_ms, peak_memory_mb = measure_inference(model, input_ids, attention_mask, device, n_runds=args.n_runs)
    print(f"avg inference time: {avg_time_ms:.2f}ms")
    print(f"peak memory: {peak_memory_mb:.2f}MB")

    results = {
        "variant": args.variant,
        "task": args.task,
        "accuracy": accuracy,
        "avg_inference_time_ms": avg_time_ms,
        "peak_memory_mb": peak_memory_mb,

        "seq_len": config["seq_len"],
        "n_runs": args.n_runs,
    }

scripts/visualise.py

Generates plots from saved evaluation results and training logs.

Reads JSON files from results/ and produces:
    - accuracy comparison bar chart across all four variants per task.
    - inference time bar chart across all four variants per task.
    - training loss curves per variant per task.

Saves plots to results/plots/ as .png and .pdf

Usage:
    python3 scripts/visualise.py --task listops
    python3 scripts/visualise.py --task text_classification
    python3 scripts/visualise.py --task integration_test
"""

import os
import sys
import json
import argparse
import matplotlib.pyplot as plt
import matplotlib
matplotlib.use("Agg") # non-interactive backend so it works without display

sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

ROOT_DIR = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
RESULTS_DIR = os.path.join(ROOT_DIR, "results")
PLOTS_DIR = os.path.join(RESULTS_DIR, "plots")

VARIANTS = ["baseline", "longformer", "linformer", "reformer"]
COLOURS = {"baseline": "blue", "longformer": "red", "linformer": "green", "reformer": "orange"}

def load_eval_results(task):
    """ Load evaluation JSON files for all variants for the given task"""
    results = {}
    for variant in VARIANTS:
        path = os.path.join(RESULTS_DIR, f"{variant}_{task}_eval.json")
        if os.path.exists(path):
            with open(path, "r") as f:
                results[variant] = json.load(f)
        else:
            print(f"warning: no eval results found for {variant} on {task} - skipping")
    return results

def load_training_logs(task):
    """ Load training log JSON files for all variants for the given task"""
    logs = {}
    for variant in VARIANTS:
        path = os.path.join(RESULTS_DIR, f"{variant}_{task}_log.json")
        if os.path.exists(path):
            with open(path, "r") as f:
                logs[variant] = json.load(f)
        else:
            print(f"warning: no training log found for {variant} on {task} - skipping")
    return logs

def save_plot(fig, name):
    """ save as both png (for dissertation doc) and pdf (for quality)"""
    os.makedirs(PLOTS_DIR, exist_ok=True)
    fig.savefig(os.path.join(PLOTS_DIR, f"{name}.png"), dpi=150, bbox_inches="tight")
    fig.savefig(os.path.join(PLOTS_DIR, f"{name}.png"), bbox_inches="tight")
    print(f"saved {name}.png and {name}.pdf")

def plot_accuracy(results, task):
    """ Bar chart comparing test accuracy across all 4 variants"""
    variants = list(results.keys())
    accuracies = [results[v]["accuracy"] * 100 for v in variants]

    fig, ax = plt.subplots()
    ax.bar(variants, accuracies)
    ax.set_xlabel("variant")
    ax.set_ylabel("test accuracy (%)")
    ax.set_title(f"accuracy comparison - {task}")

    save_plot(fig, f"accuracy_{task}")
    plt.close(fig)

def plot_inference_time(results, task):
    """Bar chart comparing avg inference time per forward pass"""
    variants = list(results.keys())
    times = [results[v]["avg_inference_time_ms"] for v in variants]

    fig, ax = plt.subplots()
    ax.bar(variants, times)
    ax.set_xlabel("variant")
    ax.set_ylabel("avg inference time (ms)")
    ax.set_title(f"inference time comparison - {task}")

    save_plot(fig, f"inference_time_{task}")
    plt.close(fig)

def plot_memory(results, task):
    """Bar chart comparing peak memory usage. Only available on CUDA - skippped on CPU/MPS"""
    variants = list(results, task)
    memory = [results[v]["peak_memory_mb"] for v in variants]

    # peak memory stats are only available on CUDA - MPS and CPU return 0
    if all(m == 0 for m in memory):
        print(f"skipping memory plot for {task} - peak memory only available on CUDA")
        return
    
    fig, ax = plt.subplots()
    ax.bar(variants, memory)
    ax.set_xlabel("variant")
    ax.set_ylabel("peak memory (MB)")
    ax.set_title(f"peak memory comparison - {task}")

    save_plot(fig, f"memory_{task}")
    plt.close(fig)

def plot_seq_len_benchmark(task=None):
    """
    Line plots showing inference time vs sequence length for all four variants.
    This directly tests the complexity claims from the papers - baseline should grow quadratically while efficient variants should grow more slowly.
    """
    path = os.path.join(RESULTS_DIR, "benchmark_seq_len.json")
    if not os.path.exists(path):
        print("no sequence length benchmark found - run scripts/benchmark_seq_len.py first")
        return

    with open(path, "r") as f:
        data = json.load(f)

    # inference time plot
    fig, ax = plt.subplots()
    for variant in VARIANTS:
        if variant not in data:
            continue
        seq_lengths = data[variant]["seq_lengths"]
        times = data[variant]["avg_time_ms"]
        ax.plot(seq_lengths, times, color=COLOURS[variant], marker="o", label=variant)

    ax.set_xlabel("sequence length")
    ax.set_ylabel("avg inference time (ms)")
    ax.set_title("inference time vs sequence length")
    ax.legend()
    save_plot(fig, "seq_len_inference_time")
    plt.close(fig)

    # memory plot - skip if all zeros (CPU/MPS)
    memory_vals = [data[v]["peak_memory_mb"] for v in VARIANTS if v in data]
    if any(m > 0 for ms in memory_vals for m in ms):
        fig, ax = plt.subplots()
        for variant in VARIANTS:
            if variant not in data:
                continue
            seq_lengths = data[variant]["seq_lengths"]
            memory = data[variant]["peak_memory_mb"]
            ax.plot(seq_lengths, memory, color=COLOURS[variant], marker="o", label=variant)

        ax.set_xlabel("sequence length")
        ax.set_ylabel("peak_memory (MB)")
        ax.set_title("peak memory vs sequence length")
        ax.legend()
        save_plot(fig, "seq_len_memory")
        plt.close(fig)
    else:
        print("skipping memory vs seq_len plot - peak memory only available on CUDA")

def plot_training_curves(logs, task):
    """
    Training loss and accuracy curves for all variants on the same axes.
        Solid lines = train
        Dashed lines = test
    Shows overfitting at a glance.
    """
    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 4))

    for variant, log in logs.items():
        epochs = [e["epoch"] for e in log]
        train_loss = [e["train_loss"] for e in log]
        test_loss = [e["test_loss"] for e in log]
        train_acc = [e["train_acc"] * 100 for e in log]
        test_acc = [e["test_acc"] * 100 for e in log]

        # same colour per variant
        # solid for train / dashed for test - same for both subplots
        ax1.plot(epochs, train_loss, color=COLOURS[variant], label=f"{variant} train")
        ax1.plot(epochs, test_loss, color=COLOURS[variant], linestyle="--", label=f"{variant} test")

        ax2.plot(epochs, train_acc, color=COLOURS[variant], label=f"{variant} train")
        ax2.plot(epochs, test_acc, color=COLOURS[variant], linestyle="--", label=f"{variant} test")
    
    ax1.set_xlabel("epoch")
    ax1.set_ylabel("loss")
    ax1.set_title(f"training loss - {task}")
    ax1.legend(fontsize=7)

    ax2.set_xlabel("epoch")
    ax2.set_ylabel("accuracy (%)")
    ax2.set_title(f"training accuracy - {task}")
    ax2.legend(fontsize=7)

    fig.tight_layout()
    save_plot(fig, f"training_curves_{task}")
    plt.close(fig)



def main():

    parser = argparse.ArgumentParser()
    parser.add_argument("--task", required=True, choices=["listops", "text_classification", "integration_test"])
    args = parser.parse_args()

    print(f"generating plots for {args.task}...\n")

    # sequence length benchmark - independent of task from benchmark_seq_len.py
    plot_seq_len_benchmark()

    # training curves only need training logs
    logs = load_training_logs(args.task)
    if logs:
        plot_training_curves(logs, args.task)

    #accuracy, timing and memory need evaluate.py to have been run first
    results = load_eval_results(args.task)
    if results:
        plot_accuracy(results, args.task)
        plot_inference_time(results, args.task)
        plot_memory(results, args.task)
    else:
        print("no eval results found - run scripts/evaluate.py first for accuracy and timing plots")

    print(f"\nall plots saved to results/plots/")

scripts/benchmark_seq_len.py

Benchmarks inference time and peak memory for all four attention variants across multiple sequence lengths using random dummy input.

No trained model or real data needed - just measures how long a forward pass takes and how much memory
it uses at each sequence length. This directly tests the complexity claims from the papers:
- baseline -> O(n^2)
- longformer -> O(n * window_size)
- linformer -> O(n * proj_dim)
- reformer -> O(n long n)

The Reformer paper [X] Kitaev et al. (2020) Figure 5 shows LSH only beats full attention beyond ~2048 tokens
    - this benchmark replicates that experiment.

Results are saved to results/benchmark_seq_len.json for plotting with visualise.py

Usage: python3 scripts/benchmark_seq_len.py
"""

import os
import sys
import json

RESULTS_DIR = os.path.join(ROOT_DIR, "results")

VARIANTS = ["baseline", "longformer", "linformer", "reformer"]

# sequence lengths to benchmark - covers short to full LRA lengths
SEQ_LENGTHS = [128, 256, 512, 1024, 2048]

# small model config - same across all variants so compairson is fair
# not using full d_model=512 here as baseline at seq_len=2048 as OOM's Macbook Pro M3
BENCH_CONFIG = {
    "d_model" : 128,
    "num_heads" : 4,
    "num_layers" : 2,
    "d_ff" : 512,
    "dropout" : 0.0, # no dropout during benchmarking
    "vocab_size" : 100,
    "num_classes" : 10,
    "batch_size" : 1,
    "longformer_window_size" : 64,
    "linformer_proj_dim" : 64,
    "reformer_n_rounds" : 4,
    "reformer_chunk_size" : 32,
    "reformer_n_buckets" : 8,
}

N_RUNS = 20 # number of forward passes to avg over

def benchmark_variant(variant, seq_len, device, n_runs=N_RUNS):
    """
    Measure average inference time and peak memory for noe variant at on sequence length.

    Runs a warmup pass first to avoid measuring JIT/compilation overhead then averages over n_runs forward passes.

    Args:
        variant (str): attention variant name
        seq_len (int): sequence length to benchmark
        device: torch device.
        n_runs (int): number of forward passes to average

    Returns:
        avg_time_ms (float): average forward pass time in milliseconds
        peak_memory_mb (float): peak memory in MB (CUDA only, 0 on CPU/MPS)
    """

    # linformer needs seq_len at construction time
    config = dict(BENCH_CONFIG)
    config["seq_len"] = seq_len

    model = build_model(config, variant)
    model = model.to(device)
    model.eval()

    # random dummpy input - we only care about time and memory, not accuracy
    input_ids = torch.randint(0, config["vocab_size"], (1, seq_len)).to(device)
    attention_mask = torch.ones(1, seq_len, dtype=torch.long).to(device)

    # warmup pass
    with torch.no_grad():
        _ = model(input_ids, attention_mask=attention_mask)

    if device.type == "cuda":
        torch.cuda.reset_peak_memory_stats()

    times = []
    with torch.no_grad():
        for _ in range(n_runs):
            if device.type == "cuda":
                start = torch.cuda.Event(enable_timing=True)
                end = torch.cuda.Event(enable_timing=True)
                start.record()
                _ = model(input_ids, attention_mask=attention_mask)
                end.record()
                torch.cuda.synchronize()
                times.append(start.elapsed_time(end))
            else:
                t0 = time.perf_counter()
                _ = model(input_ids, attention_mask=attention_mask)
                t1 = time.perf_counter()
                times.append((t1 - t0) * 1000)

    avg_time_ms = sum(times) / len(times)

    if device.type == "cuda":
        peak_memory_mb = torch.cuda.max_memory_allocated() / 1024 ** 2
    else:
        # MPS and CPU dont expose peak memory stats same way
        peak_memory_mb = 0.0

    return avg_time_ms, peak_memory_mb


def main():
    device = get_device()
    print(f"using device: {device}")
    print(f"benchmarking {len(VARIANTS)} variants across {len(SEQ_LENGTHS)} sequence lengths... \n")

    results = {}

    for variant in VARIANTS:
        results[variant] = {"seq_lengths": [], "avg_time_ms": [], "peak_memory_mb": []}

        for seq_len in SEQ_LENGTHS:
            print(f"{variant} - seq_len={seq_len}...", end=" ", flush=True)

            try:
                avg_time, peak_mem = benchmark_variant(variant, seq_len, device)
                results[variant]["seq_lengths"].append(seq_len)
                results[variant]["avg_time_ms"].append(avg_time)
                results[variant]["peak_memory_mb"].append(peak_mem)
                print(f"{avg_time:.1f}ms")

            except RuntimeError as e:
                # OOM or other runtime error - skip this seq_len for this variant
                print(f"skipped ({e})")

    os.makedirs(RESULTS_DIR, exist_ok=True)
    out_path = os.path.join(RESULTS_DIR, "benchmark_seq_len.json")
    with open(out_path, "w") as f:
        json.dump(results, f, indent=2)

    print(f"\nbenchmark complete. results saved to {out_path}")
    print("run visualise.py to generate plots from these results")

scripts/preprocess_listops.py

Converts the ListOps TSV files generated by data/listops.py into PyTorch TensorDatasets saved as .pt files
in data/listops/. These are what train.py loads at runtime.

Usage:
    python3 scripts/preprocess_listops.py
"""

import os
import sys
import csv
import torch
from torch.utils.data import TensorDataset

ROOT_DIR = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
RAW_DIR = os.path.join(ROOT_DIR, "data", "listops_raw")
OUT_DIR = os.path.join(ROOT_DIR, "data", "listops")

# must match seq_len in configs/listops.yaml
SEQ_LEN = 2048

VOCAB = {
    "[PAD]": 0, "[UNK]": 1, "(": 2, ")": 3,
    "MIN": 4, "MAX": 5, "MED": 6, "FIRST": 7, "LAST": 8, "SUM": 9,
    "0": 10, "1": 11, "2": 12, "3": 13, "4": 14, "5": 15,  "6": 16, "7": 17, "8": 18, "9": 19,
}

def tokenise(sequence):
    tokens = [VOCAB.get(t, VOCAB["[UNK]"]) for t in sequence.strip().split()]
    tokens = tokens[:SEQ_LEN]
    pad_len = SEQ_LEN - len(tokens)
    mask = [1] * len(tokens) + [0] * pad_len
    tokens = tokens + [VOCAB["[PAD]"]] * pad_len
    return tokens, mask

def parse_tsv(tsv_path):
    input_ids_list, labels_list, masks_list = [], [], []
    with open(tsv_path, "r") as f:
        reader = csv.DictReader(f, delimiter="\t")
        for row in reader:
            tokens, mask = tokenise(row["Source"])
            label = int(row["Target"])
            input_ids_list.append(tokens)
            labels_list.append(label)
            masks_list.append(mask)
    return TensorDataset(
        torch.tensor(input_ids_list, dtype=torch.long),
        torch.tensor(labels_list, dtype=torch.long),
        torch.tensor(masks_list, dtype=torch.long),
    )

def main():
    os.makedirs(OUT_DIR, exist_ok=True)

    for split in ["train", "test"]:
        tsv_path = os.path.join(RAW_DIR, f"basic_{split}.tsv")
        if not os.path.exists(tsv_path):
            raise FileNotFoundError(f"could not find {tsv_path}. run data/listops.py first")
        
        print(f"parsing {split} split..")
        dataset = parse_tsv(tsv_path)
        save_path = os.path.join(OUT_DIR, f"{split}.pt")
        torch.save(dataset, save_path)
        print(f"saved {len(dataset)} examples to {save_path}")

    print("\ndone - ListOps data ready at data/listops/")


if __name__ == "__main__":
    main()
